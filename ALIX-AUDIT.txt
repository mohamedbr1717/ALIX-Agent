def dangerous(command):

    command_lower = command.lower()

    return any(
        pattern.lower() in command_lower
        for pattern in DANGEROUS_PATTERNS
    )

def list_files(path="."):

    try:
        p = Path(path).expanduser()

        if not p.exists():
            return {
                "success": False,
                "error": "المسار غير موجود"
            }

        if p.is_file():
            return {
                "success": True,
                "type": "file",
                "path": str(p)
            }

        items = []

        for item in sorted(p.iterdir()):

            items.append({
                "name": item.name,
                "path": str(item),
                "type": "directory" if item.is_dir() else "file"
            })

        return {
            "success": True,
            "path": str(p),
            "items": items[:300]
        }

    except Exception as e:

        return {
            "success": False,
            "error": str(e)
        }

def read_file(path, start_line=None, end_line=None):

    try:

        p = Path(path).expanduser()

        if not p.exists():
            return {
                "success": False,
                "error": "الملف غير موجود"
            }

        if not p.is_file():
            return {
                "success": False,
                "error": "المسار ليس ملفًا"
            }

        size = p.stat().st_size

        if size > 2_000_000:
            return {
                "success": False,
                "error": "الملف أكبر من 2MB"
            }

        content = p.read_text(
            encoding="utf-8",
            errors="replace"
        )

        lines = content.splitlines()

        total_lines = len(lines)

        # ---------------------------------------------
        # Optional line-range reading
        # ---------------------------------------------

        if start_line is not None or end_line is not None:

            try:
                start = int(start_line) if start_line is not None else 1
                end = int(end_line) if end_line is not None else total_lines

            except (TypeError, ValueError):
                return {
                    "success": False,
                    "error": "start_line/end_line يجب أن يكونا أرقامًا"
                }

            if start < 1:
                start = 1

            if end < start:
                return {
                    "success": False,
                    "error": "end_line يجب أن يكون أكبر أو مساويًا لـ start_line"
                }

            selected = lines[start - 1:end]

            return {
                "success": True,
                "path": str(p),
                "content": "\n".join(selected),
                "start_line": start,
                "end_line": min(end, total_lines),
                "total_lines": total_lines
            }

        # ---------------------------------------------
        # Full file read
        # ---------------------------------------------

        return {
            "success": True,
            "path": str(p),
            "content": content[:40000],
            "start_line": 1,
            "end_line": min(total_lines, 40000),
            "total_lines": total_lines
        }

    except Exception as e:

        return {
            "success": False,
            "error": str(e)
        }

def search_files(path, pattern):

    try:

        root = Path(path).expanduser()

        if not root.exists():
            return {
                "success": False,
                "error": "المجلد غير موجود"
            }

        matches = []

        for p in root.rglob("*"):

            if len(matches) >= 200:
                break

            if p.is_file() and pattern.lower() in p.name.lower():

                matches.append(str(p))

        return {
            "success": True,
            "matches": matches
        }

    except Exception as e:

        return {
            "success": False,
            "error": str(e)
        }

    except Exception as e:

        return {
            "success": False,
            "error": str(e)

def run_terminal(command):

    print("\n╭────────────────────────────────────╮")
    print("│       TERMINAL EXECUTION           │")
    print("╰────────────────────────────────────╯")

    print(command)

    if dangerous(command):

        print("\n🚨 تحذير: هذا الأمر مصنف كأمر خطير.")

        answer = input(
            "اكتب ALIX-YES للتأكيد: "
        ).strip()

        if answer != "ALIX-YES":

            return {
                "success": False,
                "error": "تم رفض الأمر الخطير"
            }

    else:

        answer = input(
            "هل تسمح لـ ALIX بتنفيذ الأمر؟ [y/N]: "
        ).strip().lower()

        if answer != "y":

            return {
                "success": False,
                "error": "المستخدم رفض التنفيذ"
            }

    try:

        result = subprocess.run(
            command,
            shell=True,
            capture_output=True,
            text=True,
            timeout=120
        )

        return {
            "success": result.returncode == 0,
            "return_code": result.returncode,
            "stdout": result.stdout[:20000],
            "stderr": result.stderr[:10000]
        }

    except subprocess.TimeoutExpired:

        return {
            "success": False,
            "error": "انتهت مهلة التنفيذ"
        }

    except Exception as e:

        return {
            "success": False,
            "error": str(e)
        }

def execute_tool(name, arguments):

    if name == "list_files":
        return list_files(
            arguments.get("path", ".")
        )

    if name == "read_file":
        return read_file(
            arguments["path"],
            arguments.get("start_line"),
            arguments.get("end_line")
        )

    if name == "search_files":
        return search_files(
            arguments["path"],
            arguments["pattern"]
        )

    if name == "write_file":
        return write_file(
            arguments["path"],
            arguments["content"]
        )

    if name == "save_fact":
        return {
            "success": memory_manager.add_fact(
                arguments["key"],
                arguments["value"],
                REAL_MEMORY_FILE
            ),
            "key": arguments["key"]
        }

    if name == "get_fact":
        value = memory_manager.get_fact(
            arguments["key"],
            REAL_MEMORY_FILE
        )

        return {
            "success": value is not None,
            "key": arguments["key"],
            "value": value
        }

    if name == "save_preference":
        return {
            "success": memory_manager.add_preference(
                arguments["key"],
                arguments["value"],
                REAL_MEMORY_FILE
            ),
            "key": arguments["key"]
        }

    if name == "get_preference":
        value = memory_manager.get_preference(
            arguments["key"],
            REAL_MEMORY_FILE
        )

        return {
            "success": value is not None,
            "key": arguments["key"],
            "value": value
        }

    if name == "memory_summary":
        return {
            "success": True,
            "summary": memory_manager.get_summary(
                REAL_MEMORY_FILE
            )
        }

    if name == "run_terminal":
        return run_terminal(
            arguments["command"]
        )

    return {
        "success": False,
        "error": f"Unknown tool: {name}"
    }

    if any(word in text for word in write_words):
        return "file_write"

    # -----------------------------
    # Terminal
    # -----------------------------

    terminal_words = [
        "نفذ أمر",
        "شغل الأمر",
        "شغّل الأمر",
        "terminal",
        "ترمينال",
        "bash",
        "shell",
        "command",
    ]

    if any(word in text for word in terminal_words):
        return "terminal"

    # -----------------------------
    # Investigation / debugging
    # -----------------------------

    investigation_words = [
        "اكتشف الخطأ",
        "افحص المشروع",
        "حلل المشروع",
        "شخّص الخطأ",
        "لماذا لا يعمل",
        "ما سبب الخطأ",
        "debug",
        "diagnose",
        "inspect project",
    ]

    if any(word in text for word in investigation_words):
        return "investigation"

    # -----------------------------
    # Complex planning
    # -----------------------------

    planning_words = [
        "خطط",
        "خطة",
        "خطوة بخطوة",
        "نفذ المشروع",
        "ابن المشروع",
        "طوّر المشروع",
        "طور المشروع",
        "أضف نظام",
        "صمم نظام",
        "build",
        "develop",
        "implement",
        "architecture",
    ]

    if any(word in text for word in planning_words):
        return "planning"

    return "chat"


def intelligence_layer_1(user_input):
    """
    ALIX V4 Intelligence Layer 1.

    يحلل الطلب فقط ويحدد:
    - intent
    - هل يحتاج تخطيطًا
    - هل يحتاج أدوات

    لا ينفذ أدوات ولا Terminal.
    """

    intent = classify_intent(user_input)

    planning_intents = {
        "planning",
        "investigation",
    }

    tool_intents = {
        "memory",
        "file_read",
        "file_search",
        "file_write",
        "terminal",
        "investigation",
    }

    return {
        "intent": intent,
        "needs_plan": intent in planning_intents,
        "needs_tools": intent in tool_intents,
    }

def decompose_task(user_input, plan=None):
    """
    يحول المهمة إلى خطوات منطقية قابلة للتنفيذ.

    لا يستخدم Terminal.
    لا ينفذ أدوات.
    لا يعدل الملفات.

    إذا كان Planner موجودًا، يستخدم خطته كمرجع.
    """

    if not isinstance(user_input, str):
        return []

    text = user_input.strip()

    if not text:
        return []

    steps = []

    if isinstance(plan, list):
        for item in plan:
            if isinstance(item, str) and item.strip():
                steps.append(item.strip())

    if not steps:
        intent = classify_intent(text)

        if intent == "investigation":
            steps = [
                "جمع الأدلة من المشروع",
                "تحديد مصدر المشكلة",
                "اقتراح الإصلاح",
                "التحقق من النتيجة",
            ]

        elif intent == "file_read":
            steps = [
                "تحديد الملف المطلوب",
                "قراءة الملف",
                "عرض المعلومات المطلوبة",
            ]

        elif intent == "file_search":
            steps = [
                "تحديد نطاق البحث",
                "البحث عن الملفات المطلوبة",
                "عرض النتائج",
            ]

        elif intent == "file_write":
            steps = [
                "فهم التعديل المطلوب",
                "تحديد الملفات المتأثرة",
                "طلب الموافقة قبل الكتابة",
                "تنفيذ التعديل",
                "التحقق من النتيجة",
            ]

        elif intent == "terminal":
            steps = [
                "فهم الأمر المطلوب",
                "طلب الموافقة قبل التنفيذ",
                "تنفيذ الأمر",
                "التحقق من النتيجة",
            ]

        elif intent == "memory":
            steps = [
                "تحديد نوع عملية الذاكرة",
                "استخدام أداة الذاكرة المناسبة",
                "عرض النتيجة",
            ]

        else:
            steps = [
                "فهم طلب المستخدم",
                "تقديم الإجابة المناسبة",
            ]

    return steps


def verify_task_result(user_input, result, intent=None):
    """
    فحص بسيط للنتيجة النهائية.

    لا ينفذ أدوات.
    لا يستخدم Terminal.
    لا يغير النتيجة.

    Returns:
        dict يحتوي على:
        success
        reason
        needs_followup
    """

    if intent is None:
        intent = classify_intent(user_input)

    if result is None:
        return {
            "success": False,
            "reason": "لم يتم الحصول على نتيجة.",
            "needs_followup": True,
        }

    if not isinstance(result, str):
        result_text = str(result)
    else:
        result_text = result

    result_text = result_text.strip()

    if not result_text:
        return {
            "success": False,
            "reason": "النتيجة فارغة.",
            "needs_followup": True,
        }

    error_markers = [
        "❌",
        "Traceback",
        "Exception:",
        "Error:",
    ]

    has_error = any(
        marker in result_text
        for marker in error_markers
    )

    if has_error:
        return {
            "success": False,
            "reason": "النتيجة تحتوي على مؤشر خطأ.",
            "needs_followup": True,
        }

    return {
        "success": True,
        "reason": "تم الحصول على نتيجة غير فارغة ولم يظهر مؤشر خطأ واضح.",
        "needs_followup": False,
    }

def orchestrate_task(user_input):
    """
    يبني سياق تنفيذ موحد للمهمة.

    هذه الطبقة لا تنفذ أدوات.
    لا تستخدم Terminal.
    لا تعدل الملفات.
    لا تتجاوز نظام الموافقة.

    Pipeline:

        User Request
             ↓
        Classification
             ↓
        Planning (when needed)
             ↓
        Task Decomposition
             ↓
        Execution Guidance
             ↓
        Verification Guidance
    """

    if not isinstance(user_input, str):
        return {
            "success": False,
            "intent": "chat",
            "needs_tools": False,
            "needs_plan": False,
            "plan": [],
            "steps": [],
            "verification_required": False,
            "reason": "طلب المستخدم غير صالح.",
        }

    text = user_input.strip()

    if not text:
        return {
            "success": False,
            "intent": "chat",
            "needs_tools": False,
            "needs_plan": False,
            "plan": [],
            "steps": [],
            "verification_required": False,
            "reason": "الطلب فارغ.",
        }

    # --------------------------------------------------------
    # Layer 1 — Classification
    # --------------------------------------------------------

    intelligence = intelligence_layer_1(text)

    intent = intelligence.get("intent", "chat")
    needs_tools = bool(
        intelligence.get("needs_tools", False)
    )
    needs_plan = bool(
        intelligence.get("needs_plan", False)
    )

    # --------------------------------------------------------
    # Layer 2 — Planning
    # --------------------------------------------------------

    plan = []

    if needs_plan:
        try:
            generated_plan = create_plan(text)

            if isinstance(generated_plan, list):
                plan = [
                    item.strip()
                    for item in generated_plan
                    if isinstance(item, str)
                    and item.strip()
                ]

        except Exception:
            plan = []

    # --------------------------------------------------------
    # Layer 2 — Task Decomposition
    # --------------------------------------------------------

    steps = decompose_task(
        text,
        plan if plan else None
    )

    # --------------------------------------------------------
    # Verification policy
    # --------------------------------------------------------

    verification_required = bool(
        needs_tools
        or needs_plan
        or intent in {
            "file_read",
            "file_search",
            "file_write",
            "investigation",
            "memory",
            "terminal",
        }
    )

    # --------------------------------------------------------
    # Execution guidance
    # --------------------------------------------------------

    if needs_tools:
        execution_mode = "tool_required"
    else:
        execution_mode = "model_response"

    # --------------------------------------------------------
    # Safety guidance
    # --------------------------------------------------------

    approval_required = intent in {
        "file_write",
        "terminal",
    }

    return {
        "success": True,
        "intent": intent,
        "needs_tools": needs_tools,
        "needs_plan": needs_plan,
        "plan": plan,
        "steps": steps,
        "execution_mode": execution_mode,
        "verification_required": verification_required,
        "approval_required": approval_required,
    }

def resolve_evidence_targets(targets):
    """
    يحول أهداف التحقيق المنطقية إلى ملفات فعلية.

    Read-only:
    - لا Terminal
    - لا كتابة ملفات
    - لا تعديل المشروع
    """

    if not isinstance(targets, list):
        return []

    resolved = []

    function_to_file = {
        "run_agent": "agent.py",
        "collect_evidence": "agent.py",
        "analyze_evidence": "agent.py",
        "classify_evidence": "agent.py",
        "filter_evidence_context": "agent.py",
        "investigate_project": "agent.py",
        "orchestrate_task": "agent.py",
        "execute_tool": "agent.py",
        "create_plan": "agent.py",
        "classify_intent": "agent.py",
        "memory_manager": "memory_manager.py",
    }

    for target in targets:

        if not isinstance(target, str):
            continue

        target = target.strip()

        if not target:
            continue

        mapped = function_to_file.get(target)

        if mapped:
            target = mapped

        elif target == "PROJECT":

            for project_file in [
                "agent.py",
                "memory_manager.py",
            ]:
                if project_file not in resolved:
                    resolved.append(project_file)

            continue

        elif not target.endswith(".py"):

            candidate = f"{target}.py"

            if (BASE_DIR / candidate).exists():
                target = candidate

        if target not in resolved:
            resolved.append(target)

    return resolved


def collect_evidence(targets):
    """
    يجمع أدلة آمنة من ملفات المشروع.

    القواعد:
    - قراءة فقط.
    - لا Terminal.
    - لا كتابة.
    - لا تعديل.
    - كل دليل يحتوي على مصدر واضح.
    """

    if not isinstance(targets, list):
        return {
            "success": False,
            "evidence": [],
            "error": "targets يجب أن تكون قائمة"
        }

    evidence = []

    for target in targets:

        if not isinstance(target, str):
            continue

        target = target.strip()

        if not target:
            continue

        # لا نحاول تنفيذ أي شيء.
        # نقرأ الملفات الموجودة فقط.
        path = Path(target).expanduser()

        if not path.is_absolute():
            path = BASE_DIR / path

        if not path.exists():
            evidence.append({
                "type": "missing_target",
                "target": target,
                "success": False,
                "message": "الهدف غير موجود"
            })
            continue

        if not path.is_file():
            evidence.append({
                "type": "non_file_target",
                "target": target,
                "success": False,
                "message": "الهدف ليس ملفًا"
            })
            continue

        result = read_file(str(path))

        if not result.get("success"):
            evidence.append({
                "type": "read_error",
                "target": target,
                "success": False,
                "message": result.get(
                    "error",
                    "فشل في قراءة الملف"
                )
            })
            continue

        content = result.get("content", "")

        evidence.append({
            "type": "file_content",
            "target": target,
            "path": result.get("path"),
            "success": True,
            "content": content,
            "total_lines": result.get("total_lines"),
        })

    return {
        "success": True,
        "count": len(evidence),
        "evidence": evidence,
    }

def analyze_evidence(evidence_result):
    """
    يحلل الأدلة التي تم جمعها مسبقًا.

    لا يقرأ ملفات جديدة.
    لا يستخدم Terminal.
    لا يكتب الملفات.
    لا يعدل المشروع.

    النتيجة:
        file
        line
        evidence
        reason
        confidence
    """

    if not isinstance(evidence_result, dict):
        return {
            "success": False,
            "findings": [],
            "error": "evidence_result يجب أن يكون dict"
        }

    if not evidence_result.get("success"):
        return {
            "success": False,
            "findings": [],
            "error": "فشل جمع الأدلة"
        }

    findings = []

    evidence_items = evidence_result.get(
        "evidence",
        []
    )

    if not isinstance(evidence_items, list):
        return {
            "success": False,
            "findings": [],
            "error": "evidence يجب أن تكون قائمة"
        }

    suspicious_patterns = [
        (
            "SyntaxError",
            "وجود مؤشر SyntaxError في الملف",
            "high"
        ),
        (
            "Traceback",
            "وجود Traceback في محتوى الملف",
            "high"
        ),
        (
            "TODO",
            "وجود TODO قد يشير إلى عمل غير مكتمل",
            "low"
        ),
        (
            "FIXME",
            "وجود FIXME قد يشير إلى مشكلة معروفة",
            "medium"
        ),
        (
            "except Exception:",
            "استخدام معالجة عامة للاستثناءات؛ قد تخفي أخطاء حقيقية",
            "low"
        ),
    ]

    for item in evidence_items:

        if not isinstance(item, dict):
            continue

        if item.get("type") != "file_content":
            continue

        if not item.get("success"):
            continue

        file_path = item.get("path") or item.get("target")
        content = item.get("content", "")

        if not isinstance(content, str):
            continue

        lines = content.splitlines()

        for line_number, line in enumerate(lines, 1):

            for pattern, reason, confidence in suspicious_patterns:

                if pattern.lower() in line.lower():

                    findings.append({
                        "file": file_path,
                        "line": line_number,
                        "evidence": line.strip(),
                        "pattern": pattern,
                        "reason": reason,
                        "confidence": confidence,
                    })

    return {
        "success": True,
        "count": len(findings),
        "findings": findings,
    }

def classify_evidence(analysis_result):
    """
    يصنف Findings الموجودة مسبقًا.

    التصنيفات:
        confirmed
        suspicious
        informational

    لا يقرأ ملفات جديدة.
    لا يستخدم Terminal.
    لا يعدل المشروع.
    """

    if not isinstance(analysis_result, dict):
        return {
            "success": False,
            "findings": [],
            "error": "analysis_result يجب أن يكون dict"
        }

    if not analysis_result.get("success"):
        return {
            "success": False,
            "findings": [],
            "error": "تحليل الأدلة غير ناجح"
        }

    findings = analysis_result.get("findings", [])

    if not isinstance(findings, list):
        return {
            "success": False,
            "findings": [],
            "error": "findings يجب أن تكون قائمة"
        }

    classified = []

    for finding in findings:

        if not isinstance(finding, dict):
            continue

        item = dict(finding)

        pattern = str(
            item.get("pattern", "")
        ).lower()

        evidence = str(
            item.get("evidence", "")
        ).strip()

        confidence = str(
            item.get("confidence", "")
        ).lower()

        # ----------------------------------------------------
        # Default classification
        # ----------------------------------------------------

        classification = "informational"
        reason = "الدليل لا يثبت وجود خطأ بحد ذاته."

        # ----------------------------------------------------
        # Explicit error indicators
        # ----------------------------------------------------

        if pattern == "syntaxerror":
            classification = "suspicious"
            reason = (
                "تم العثور على كلمة SyntaxError، "
                "لكن وجودها داخل النص لا يثبت وجود خطأ نحوي."
            )

        elif pattern == "traceback":
            classification = "suspicious"
            reason = (
                "تم العثور على كلمة Traceback، "
                "ويجب التحقق من السياق لمعرفة هل هي نتيجة خطأ فعلية."
            )

        elif pattern == "fixme":
            classification = "suspicious"
            reason = (
                "FIXME يشير إلى نقطة تحتاج مراجعة، "
                "لكنه لا يثبت وجود خلل."
            )

        elif pattern == "todo":
            classification = "informational"
            reason = (
                "TODO يشير إلى عمل مستقبلي أو غير مكتمل، "
                "وليس بالضرورة خطأ."
            )

        elif pattern == "except exception:":
            classification = "informational"
            reason = (
                "المعالجة العامة للاستثناءات قد تكون مقصودة، "
                "لكنها تستحق المراجعة حسب السياق."
            )

        # ----------------------------------------------------
        # Strong evidence requires explicit runtime/error data.
        # The current Layer 5 engine does not execute tests,
        # therefore it must NOT claim a confirmed bug.
        # ----------------------------------------------------

        if confidence == "high":
            if classification == "confirmed":
                pass
            else:
                classification = "suspicious"

        item["classification"] = classification
        item["classification_reason"] = reason

        classified.append(item)

    counts = {
        "confirmed": 0,
        "suspicious": 0,
        "informational": 0,
    }

    for item in classified:
        category = item.get("classification")

        if category in counts:
            counts[category] += 1

    return {
        "success": True,
        "count": len(classified),
        "counts": counts,
        "findings": classified,
    }

def filter_evidence_context(classified_result):
    """
    Reduce false positives in Layer 5 evidence.

    Read-only.
    No Terminal.
    No file modification.
    """

    if not isinstance(classified_result, dict):
        return {
            "success": False,
            "findings": [],
            "error": "classified_result must be dict"
        }

    if not classified_result.get("success"):
        return {
            "success": False,
            "findings": [],
            "error": "classification failed"
        }

    findings = classified_result.get("findings", [])

    if not isinstance(findings, list):
        return {
            "success": False,
            "findings": [],
            "error": "findings must be a list"
        }

    filtered = []

    for finding in findings:

        if not isinstance(finding, dict):
            continue

        item = dict(finding)

        evidence = str(item.get("evidence", ""))
        pattern = str(item.get("pattern", "")).lower()

        context = "code"

        if evidence.lstrip().startswith("#"):
            context = "comment"

        elif "assert " in evidence or "print(" in evidence:
            context = "test_or_diagnostic"

        elif "TODO" in evidence or "FIXME" in evidence:
            context = "annotation"

        elif pattern in ("traceback", "syntaxerror"):
            context = "error_indicator"

        if pattern in ("traceback", "syntaxerror", "fixme", "todo"):
            if context in (
                "comment",
                "test_or_diagnostic",
                "annotation"
            ):
                item["classification"] = "informational"
                item["confidence"] = "low"
                item["classification_reason"] = (
                    "المؤشر ظهر في سياق نصي أو اختباري، "
                    "وليس دليلًا مباشرًا على Bug."
                )

        if item.get("classification") == "confirmed":
            item["classification"] = "suspicious"
            item["confidence"] = "low"
            item["classification_reason"] = (
                "لا يتم تأكيد Bug دون تحقق مستقل."
            )

        item["context"] = context

        filtered.append(item)

    counts = {
        "confirmed": 0,
        "suspicious": 0,
        "informational": 0
    }

    for item in filtered:
        category = item.get("classification")

        if category in counts:
            counts[category] += 1

    return {
        "success": True,
        "count": len(filtered),
        "counts": counts,
        "findings": filtered
    }

def build_agent_verification_context(diagnosis_result):
    """
    يحول نتيجة Layer 6 إلى سياق آمن يمكن للـAgent فهمه.

    Read-only.
    No Terminal.
    No file modification.
    لا يؤكد وجود Bug.
    """

    bridge = build_verification_bridge(diagnosis_result)

    if not bridge.get("success"):
        return {
            "success": False,
            "context": "",
            "error": bridge.get(
                "error",
                "فشل بناء Verification Bridge"
            )
        }

    items = bridge.get("verification_items", [])

    lines = [
        "VERIFICATION STATUS: UNVERIFIED",
        "RULE: لا تعتبر أي فرضية Bug مؤكدًا بدون دليل مستقل.",
        "",
        "VERIFICATION REQUIREMENTS:"
    ]

    for item in items:

        if not isinstance(item, dict):
            continue

        item_id = item.get("id", "?")
        hypothesis = str(
            item.get("hypothesis", "")
        ).strip()

        required = str(
            item.get("required_evidence", "")
        ).strip()

        method = str(
            item.get("verification_method", "")
        ).strip()

        status = str(
            item.get("status", "unverified")
        ).strip()

        lines.extend([
            f"[{item_id}] Hypothesis: {hypothesis}",
            f"    Required evidence: {required}",
            f"    Verification method: {method}",
            f"    Status: {status}",
            "    Confirmed bug: NO",
            ""
        ])

    return {
        "success": True,
        "context": "\n".join(lines).strip(),
        "verification_items": items,
        "safety": bridge.get("safety", {}),
        "policy": bridge.get("policy", {}),
    }

READ_FILE_INTENT = [
    r"\bاقرأ\s+(?:ملف\s+)?[\w./-]+",
    r"\bقراءة\s+(?:ملف\s+)?[\w./-]+",
    r"\bاعرض\s+(?:محتوى\s+)?(?:ملف\s+)?[\w./-]+",
    r"\bshow\s+(?:the\s+)?(?:file\s+)?[\w./-]+",
    r"\bread\s+(?:the\s+)?(?:file\s+)?[\w./-]+",
]

def is_file_read_request(user_input):
    """Detect explicit requests to read/display a file."""
    if not isinstance(user_input, str):
        return False

    text = user_input.strip()

    return any(
        re.search(pattern, text, re.IGNORECASE)
        for pattern in READ_FILE_INTENT
    )

def create_plan(user_input):
    """
    Creates a simple execution plan without executing tools.

    This function is intentionally isolated from the Agentic Loop.
    It only analyzes the request and returns ordered steps.
    """

    if not isinstance(user_input, str):
        return []

    text = user_input.strip()

    if not text:
        return []

    steps = []

    # File/project inspection
    if any(word in text.lower() for word in [
        "افحص", "تحقق", "inspect", "check", "analyze", "حلل"
    ]):
        steps.append("فحص المشروع وجمع الأدلة أولًا")

    # Reading
    if any(word in text.lower() for word in [
        "اقرأ", "قراءة", "read", "عرض الملف"
    ]):
        steps.append("قراءة الملفات المطلوبة باستخدام read_file")

    if any(word in text.lower() for word in [
        "ابحث", "بحث", "search", "find"

        steps.append("البحث عن الملفات أو الأنماط المطلوبة")

    if any(word in text.lower() for word in [

    if any(word in text.lower() for word in [

    steps.append("التحقق من النتيجة قبل تقديم الإجابة النهائية")

    return steps

def detect_memory_intent(user_input):
    """
    Detect durable user preferences/facts without using Terminal.
    Returns None, or:
    {"type": "preference", "key": ..., "value": ...}
    {"type": "fact", "key": ..., "value": ...}
    """

    if not isinstance(user_input, str):
        return None

    text = user_input.strip()

    if not text:
        return None

    # Explicitly durable preference language.
    preference_patterns = [
        (r"من الآن فصاعدًا.*?أريدك أن (.+)", "user_preference"),
        (r"من الآن فصاعدًا.*?أفضل أن (.+)", "user_preference"),
        (r"دائمًا.*?أريدك أن (.+)", "user_preference"),
        (r"always.*?prefer (.+)", "user_preference"),
        (r"from now on.*?I want you to (.+)", "user_preference"),
    ]

    for pattern, key in preference_patterns:
        m = re.search(pattern, text, re.IGNORECASE)
        if m:
            value = m.group(1).strip()

            if memory_manager.contains_secret(value):
                return None

            return {
                "type": "preference",
                "key": key,
                "value": value
            }

    return None


def auto_save_memory(user_input):
    """
    Save only clearly durable preferences.
    Never saves secrets.
    """
    intent = detect_memory_intent(user_input)

    if not intent:
        return False

    if intent["type"] == "preference":
        return memory_manager.add_preference(
            intent["key"],
            intent["value"],
            REAL_MEMORY_FILE
        )

    if intent["type"] == "fact":
        return memory_manager.add_fact(
            intent["key"],
            intent["value"],
            REAL_MEMORY_FILE
        )

    return False

    """

    Layer 6 لا تقرأ ملفات جديدة.
    Layer 6 لا تستخدم Terminal.
    Layer 6 لا تعدل المشروع.

    Returns:
        success
        diagnosis
        hypotheses
        missing_evidence
        verification_plan
        safety
    """

    if not isinstance(filtered_result, dict):
        return {
            "success": False,
            "diagnosis": "مدخل Layer 6 غير صالح.",
            "hypotheses": [],
            "missing_evidence": [],
            "verification_plan": [],
            "safety": {
                "read_only": True,
                "modification_allowed": False,
                "terminal_allowed": False,
            },
        }

    if not filtered_result.get("success"):
        return {
            "success": False,
            "diagnosis": "فشل الحصول على الأدلة المفلترة.",
            "hypotheses": [],
            "missing_evidence": [],
            "verification_plan": [],
            "safety": {
                "read_only": True,
                "modification_allowed": False,
                "terminal_allowed": False,
            },
        }

    findings = filtered_result.get("findings", [])

    if not isinstance(findings, list):
        return {
            "success": False,
            "diagnosis": "findings يجب أن تكون قائمة.",
            "hypotheses": [],
            "missing_evidence": [],
            "verification_plan": [],
            "safety": {
                "read_only": True,
                "modification_allowed": False,
                "terminal_allowed": False,
            },
        }

    hypotheses = []
    missing_evidence = []
    verification_plan = []

    suspicious = []
    informational = []

    for finding in findings:

        if not isinstance(finding, dict):
            continue

        classification = str(
            finding.get("classification", "")
        ).lower()

        pattern = str(
            finding.get("pattern", "")
        )

        file_path = finding.get(
            "file",
            finding.get("path", "")
        )

        line_number = finding.get("line")

        evidence = str(
            finding.get("evidence", "")
        ).strip()

        item = {
            "file": file_path,
            "line": line_number,
            "pattern": pattern,
            "evidence": evidence,
            "classification": classification,
        }

        if classification == "suspicious":
            suspicious.append(item)

        elif classification == "informational":
            informational.append(item)

    # --------------------------------------------------------
    # Hypothesis generation
    # --------------------------------------------------------

    patterns = {
        str(item.get("pattern", "")).lower()
        for item in suspicious
    }

    if "syntaxerror" in patterns:

        hypotheses.append({
            "type": "syntax",
            "statement": (
                "قد توجد مشكلة نحوية، لكن ظهور كلمة SyntaxError "
                "داخل الملف وحده لا يثبت وجود SyntaxError فعلي."
            ),
            "confidence": "low",
            "evidence_count": sum(
                1 for item in suspicious
                if str(item.get("pattern", "")).lower()
                == "syntaxerror"
            ),
        })

        missing_evidence.append(
            "نتيجة py_compile أو traceback فعلي يثبت خطأ SyntaxError."
        )

        verification_plan.append(
            "تشغيل py_compile على الملف المتأثر للحصول على تحقق مستقل."
        )

    if "traceback" in patterns:

        hypotheses.append({
            "type": "runtime_error",
            "statement": (
                "قد يكون هناك مؤشر على خطأ وقت تشغيل، "
                "لكن وجود كلمة Traceback في النص لا يثبت أن الملف "
                "يحتوي على Traceback ناتج عن تنفيذ فعلي."
            ),
            "confidence": "low",
            "evidence_count": sum(
                1 for item in suspicious
                if str(item.get("pattern", "")).lower()
                == "traceback"
            ),
        })

        missing_evidence.append(
            "Traceback فعلي مرتبط بتنفيذ الوظيفة المتأثرة."
        )

        verification_plan.append(
            "إعادة إنتاج المشكلة أو تنفيذ اختبار الوظيفة المتأثرة."
        )

    if "fixme" in patterns:

        hypotheses.append({
            "type": "incomplete_work",
            "statement": (
                "وجود FIXME يشير إلى نقطة تستحق المراجعة، "
                "لكنه لا يثبت وجود Bug."
            ),
            "confidence": "low",
            "evidence_count": sum(
                1 for item in suspicious
                if str(item.get("pattern", "")).lower()
                == "fixme"
            ),
        })

        missing_evidence.append(
            "تحديد ما إذا كانت نقطة FIXME تؤثر فعليًا على السلوك."
        )

        verification_plan.append(
            "فحص السلوك المرتبط بنقطة FIXME قبل اقتراح تعديل."
        )

    # --------------------------------------------------------
    # No suspicious evidence
    # --------------------------------------------------------

    if not suspicious:

        diagnosis = (
            "لم يتم العثور على دليل suspicious كافٍ لبناء "
            "فرضية Bug. الأدلة الحالية معلوماتية فقط."
        )

    else:

        diagnosis = (
            f"تم العثور على {len(suspicious)} مؤشر suspicious، "
            "لكن لا يوجد دليل كافٍ لتأكيد Bug. "
            "يلزم تحقق مستقل قبل إصدار تشخيص نهائي."
        )

    # --------------------------------------------------------
    # Always require independent verification
    # --------------------------------------------------------

    if not verification_plan:
        verification_plan.append(
            "إجراء تحقق مستقل قبل اعتبار أي مشكلة Bug مؤكدة."
        )

    if not missing_evidence:
        missing_evidence.append(
            "دليل تنفيذي مستقل يؤكد المشكلة."
        )

    return {
        "success": True,
        "diagnosis": diagnosis,
        "hypotheses": hypotheses,
        "missing_evidence": missing_evidence,
        "verification_plan": verification_plan,
        "statistics": {
            "total_findings": len(findings),
            "suspicious": len(suspicious),
            "informational": len(informational),
            "confirmed": sum(
                1
                for item in findings
                if isinstance(item, dict)
                and item.get("classification") == "confirmed"
            ),
        },
        "safety": {
            "read_only": True,
            "modification_allowed": False,
            "terminal_allowed": False,
            "confirmed_bug_allowed": False,
        },
    }

# ============================================================
# ALIX V4 — LAYER 6.1 VERIFICATION BRIDGE
# ============================================================
# Converts diagnosis hypotheses into explicit verification
# requirements.
#
# Read-only.
# No Terminal.
# No file modification.
# Never confirms a bug by itself.
# ============================================================

def build_verification_bridge(diagnosis_result):
    """
    يبني جسر التحقق من تشخيص Layer 6.

    Layer 6.1 لا تنفذ الاختبارات.
    Layer 6.1 لا تستخدم Terminal.
    Layer 6.1 لا تعدل الملفات.
    Layer 6.1 لا تؤكد Bug من نفسها.

    الهدف:
        Hypothesis
             ↓
        Required Evidence
             ↓
        Verification Method
             ↓
        Confirmation Policy
    """

    if not isinstance(diagnosis_result, dict):
        return {
            "success": False,
            "verification_items": [],
            "error": "diagnosis_result يجب أن يكون dict",
            "safety": {
                "read_only": True,
                "modification_allowed": False,
                "terminal_allowed": False,
                "confirmation_allowed": False,
            },
        }

    if not diagnosis_result.get("success"):
        return {
            "success": False,
            "verification_items": [],
            "error": "نتيجة التشخيص غير ناجحة",
            "safety": {
                "read_only": True,
                "modification_allowed": False,
                "terminal_allowed": False,
                "confirmation_allowed": False,
            },
        }

    hypotheses = diagnosis_result.get("hypotheses", [])

    if not isinstance(hypotheses, list):
        return {
            "success": False,
            "verification_items": [],
            "error": "hypotheses يجب أن تكون قائمة",
            "safety": {
                "read_only": True,
                "modification_allowed": False,
                "terminal_allowed": False,
                "confirmation_allowed": False,
            },
        }

    verification_items = []

    for index, hypothesis in enumerate(hypotheses, 1):

        if not isinstance(hypothesis, dict):
            continue

        hypothesis_type = str(
            hypothesis.get("type", "unknown")
        )

        statement = str(
            hypothesis.get("statement", "")
        )

        confidence = str(
            hypothesis.get("confidence", "unknown")
        )

        if hypothesis_type == "syntax":

            required_evidence = (
                "نتيجة py_compile أو أداة تحقق نحوي مستقلة."
            )

            verification_method = (
                "Syntax verification"
            )

        elif hypothesis_type == "runtime_error":

            required_evidence = (
                "Traceback فعلي ناتج عن تنفيذ الوظيفة المتأثرة."
            )

            verification_method = (
                "Runtime reproduction"
            )

        elif hypothesis_type == "incomplete_work":

            required_evidence = (
                "دليل سلوكي يثبت أن نقطة FIXME تؤثر على الوظيفة."
            )

            verification_method = (
                "Behavioral verification"
            )

        else:

            required_evidence = (
                "دليل مستقل مرتبط مباشرة بالفرضية."
            )

            verification_method = (
                "Independent verification"
            )

        verification_items.append({
            "id": index,
            "hypothesis_type": hypothesis_type,
            "hypothesis": statement,
            "current_confidence": confidence,
            "required_evidence": required_evidence,
            "verification_method": verification_method,
            "status": "unverified",
            "can_confirm_bug": False,
        })

    # --------------------------------------------------------
    # If there are no hypotheses, still provide a safe state.
    # --------------------------------------------------------

    if not verification_items:

        verification_items.append({
            "id": 1,
            "hypothesis_type": "none",
            "hypothesis": "لا توجد فرضية قابلة للتحقق حاليًا.",
            "current_confidence": "none",
            "required_evidence": (
                "لا يوجد دليل إضافي مطلوب حتى تظهر فرضية."
            ),
            "verification_method": "none",
            "status": "not_applicable",
            "can_confirm_bug": False,
        })

    return {
        "success": True,
        "count": len(verification_items),
        "verification_items": verification_items,
        "policy": {
            "unverified_hypotheses_are_not_bugs": True,
            "independent_evidence_required": True,
            "execution_required_for_confirmation": True,
        },
        "safety": {
            "read_only": True,
            "modification_allowed": False,
            "terminal_allowed": False,
            "confirmation_allowed": False,
        },
    }

if __name__ == "__main__":
    main()

import json
import urllib.request


class LocalLLM:

    def __init__(
        self,
        url="http://127.0.0.1:8081/v1/chat/completions",
        model="Qwen3.5-4B-Instruct-Q4_K_M.gguf"
    ):
        self.url = url
        self.model = model

    def chat(self, messages, tools=None):

        payload = {
            "model": self.model,
            "messages": messages,
            "temperature": 0.2,
            "max_tokens": 300
        }

        if tools:
            payload["tools"] = tools
            payload["tool_choice"] = "auto"

        data = json.dumps(
            payload,
            ensure_ascii=False
        ).encode("utf-8")

        request = urllib.request.Request(
            self.url,
            data=data,
            headers={
                "Content-Type": "application/json"
            },
            method="POST"
        )

        with urllib.request.urlopen(
            request,
            timeout=360
        ) as response:

            result = json.loads(
                response.read().decode("utf-8")
            )

        return result["choices"][0]["message"]

from pathlib import Path
import re
import shlex


class Policy:
    """
    طبقة الأمان والصلاحيات الخاصة بـ ALIX.

    القاعدة الأساسية:
    ALIX لا يستطيع التعامل مع الملفات خارج workspace.
    """

    def __init__(self):
        self.base_dir = (
            Path.home() / "ALIX-Agent"
        ).resolve()

        self.workspace = (
            self.base_dir / "workspace"
        ).resolve()

        self.workspace.mkdir(
            parents=True,
            exist_ok=True
        )

        # الأوامر الأساسية المسموح لـ ALIX باستخدامها.
        self.allowed_commands = {
            "pwd",
            "ls",
            "cat",
            "head",
            "tail",
            "grep",
            "find",
            "wc",
            "sort",
            "uniq",
            "cut",
            "tr",
            "echo",
            "printf",
            "date",
            "whoami",
            "uname",
            "df",
            "du",
            "free",
            "file",
            "which",
            "python",
            "python3",
            "git",
        }

        # أوامر خطرة ممنوعة دائمًا.
        self.blocked_commands = {
            "rm",
            "rmdir",
            "mkfs",
            "dd",
            "chmod",
            "chown",
            "mount",
            "umount",
            "su",
            "sudo",
            "reboot",
            "shutdown",
            "poweroff",
            "kill",
            "pkill",
        }

        # أنماط خطرة حتى لو حاول النموذج إخفاء الأمر.
        self.blocked_patterns = [
            r";",
            r"&&",
            r"\|\|",
            r"\|",
            r"`",
            r"\$\(",
            r"\$\{",
            r">\s*/",
            r">>\s*/",
            r"<\s*/",
            r"\.\./",
            r"~\/\.\.",
            r"\b/etc\b",
            r"\b/proc\b",
            r"\b/sys\b",
            r"\b/dev\b",
            r"\b/data/data\b",
            r"\b/data/user\b",
            r"\bsystemctl\b",
            r"\bservice\b",
            r"\bcurl\b.*\|\s*(sh|bash)",
            r"\bwget\b.*\|\s*(sh|bash)",
        ]

    # ========================================================
    # PATH SECURITY
    # ========================================================

    def resolve_path(self, path: str) -> Path | None:
        """
        يحول المسار إلى مسار مطلق ويتأكد أنه داخل workspace.
        """

        if not isinstance(path, str):
            return None

        path = path.strip()

        if not path:
            path = "."

        try:
            candidate = Path(path)

            if not candidate.is_absolute():
                candidate = self.workspace / candidate

            candidate = candidate.resolve()

        except Exception:
            return None

        if not self.path_allowed(candidate):
            return None

        return candidate

    def path_allowed(self, path) -> bool:
        """
        يتحقق من أن المسار داخل workspace.
        """

        try:
            path = Path(path).resolve()
            workspace = self.workspace.resolve()

            return (
                path == workspace
                or workspace in path.parents
            )

        except Exception:
            return False

    # ========================================================
    # COMMAND SECURITY
    # ========================================================

    def command_allowed(self, command: str) -> bool:
        """
        يتحقق من أن أمر shell بسيط وموجود في القائمة البيضاء.
        """

        if not isinstance(command, str):
            return False

        command = command.strip()

        if not command:
            return False

        # فحص الأنماط الخطرة أولاً.
        for pattern in self.blocked_patterns:

            if re.search(
                pattern,
                command,
                flags=re.IGNORECASE
            ):
                return False

        # تحليل الأمر بدون shell expansion.
        try:
            parts = shlex.split(command)

        except ValueError:
            return False

        if not parts:
            return False

        executable = Path(parts[0]).name

        # القائمة السوداء لها الأولوية.
        if executable in self.blocked_commands:
            return False

        # يجب أن يكون الأمر في القائمة البيضاء.
        if executable not in self.allowed_commands:
            return False

        return True

    # ========================================================
    # CONFIRMATION POLICY
    # ========================================================

    def requires_confirmation(self, action: str) -> bool:
        """
        العمليات التي يجب أن يوافق عليها المستخدم.
        """

        return action in {
            "write_file",
            "create_directory",
            "delete_file",
            "run_command",
            "run_python",
            "git",
        }

    # ========================================================
    # TOOL POLICY
    # ========================================================

    def tool_allowed(self, tool_name: str) -> bool:
        """
        يتحقق من أن الأداة معروفة لـ ALIX.
        """

        return tool_name in {
            "list_files",
            "read_file",
            "write_file",
            "create_directory",
            "delete_file",
            "run_python",
            "run_command",
            "system_info",
            "git_status",
        }

        # ----------------------------------------------------

        content = message.get(
            "content",
            ""
        ) or ""

        pattern = (
            r"<tool_call>\s*"
            r"(\{.*?\})"
            r"\s*</tool_call>"
        )

        for match in re.finditer(
            pattern,
            content,
            re.DOTALL
        ):

            try:

                obj = json.loads(
                    match.group(1)
                )

                name = obj.get(
                    "name"
                )

                arguments = obj.get(
                    "arguments",
                    {}
                )

                if name:

                    calls.append(
                        (
                            f"fallback-{len(calls)}",
                            name,
                            arguments
                        )
                    )

            except Exception:

                pass

        return calls

    # ========================================================
    # Clean response
    # ========================================================

    def clean_response(
        self,
        text
    ):

        if not text:

            return ""

        text = re.sub(
            r"<think>.*?</think>",
            "",
            text,
            flags=re.DOTALL
        )

        text = re.sub(
            r"<tool_call>.*?</tool_call>",
            "",
            text,
            flags=re.DOTALL
        )

        return text.strip()

    # ========================================================
    # Main agent loop
    # ========================================================

    def run(
        self,
        user
    ):

        self.messages.append({
            "role": "user",
            "content": user
        })

        self.memory.add_history(
            "user",
            user
        )

        # ----------------------------------------------------
        # Include compact memory context
        # ----------------------------------------------------

        context = self.memory.get_context()

        if (
            context["facts"]
            or context["preferences"]
        ):

            memory_text = (
                "\nذاكرة ALIX:\n"
                + json.dumps(
                    {
                        "facts":
                            context["facts"][-10:],
                        "preferences":
                            context["preferences"][-10:]
                    },
                    ensure_ascii=False
                )
            )

            self.messages.append({
                "role": "system",
                "content": memory_text
            })

        # ----------------------------------------------------
        # Agent loop
        # ----------------------------------------------------

        for _ in range(8):

            message = self.llm.chat(
                self.messages,
                tools=TOOLS
            )

            tool_calls = self.extract_tool_calls(
                message
            )

            # ------------------------------------------------
            # Final answer
            # ------------------------------------------------

            if not tool_calls:

                answer = self.clean_response(
                    message.get(
                        "content",
                        ""
                    )
                )

                self.messages.append({
                    "role": "assistant",
                    "content": answer
                })

                if answer:

                    self.memory.add_history(
                        "assistant",
                        answer
                    )

                    return answer

                return "لم يرجع النموذج إجابة."

            # ------------------------------------------------
            # Preserve assistant tool-call message
            # ------------------------------------------------

            self.messages.append(
                message
            )

            # ------------------------------------------------
            # Execute tools
            # ------------------------------------------------

            for (
                call_id,
                name,
                arguments
            ) in tool_calls[:4]:

                result = self.execute_tool(
                    name,
                    arguments
                )

                self.messages.append({
                    "role": "tool",
                    "tool_call_id": call_id,
                    "name": name,
                    "content": json.dumps(
                        result,
                        ensure_ascii=False
                    )
                })

        return (
            "توقّف ALIX بعد الوصول إلى "
            "الحد الأقصى لخطوات المهمة."
        )

#!/usr/bin/env python3
# -*- coding: utf-8 -*-

import json
from pathlib import Path


class Memory:

    def __init__(self, path=None):

        self.path = Path(
            path or
            Path.home() /
            "ALIX-Agent" /
            "memory" /
            "memory.json"
        )

        self.path.parent.mkdir(
            parents=True,
            exist_ok=True
        )

        if not self.path.exists():

            self._save({
                "facts": [],
                "preferences": [],
                "history": []
            })

    def _load(self):

        try:

            with self.path.open(
                "r",
                encoding="utf-8"
            ) as f:

                data = json.load(f)

            if not isinstance(data, dict):
                raise ValueError("Invalid memory format")

            return data

        except Exception:

            return {
                "facts": [],
                "preferences": [],
                "history": []
            }

    def _save(self, data):

        temp = self.path.with_suffix(".tmp")

        with temp.open(
            "w",
            encoding="utf-8"
        ) as f:

            json.dump(
                data,
                f,
                ensure_ascii=False,
                indent=2
            )

        temp.replace(self.path)

    def add_fact(self, fact):

        if not fact:
            return False

        data = self._load()

        if fact not in data["facts"]:

            data["facts"].append(fact)

            self._save(data)

        return True

    def add_preference(self, preference):

        if not preference:
            return False

        data = self._load()

        if preference not in data["preferences"]:

            data["preferences"].append(
                preference
            )

            self._save(data)

        return True

    def add_history(
        self,
        role,
        content
    ):

        data = self._load()

        data["history"].append({
            "role": role,
            "content": content
        })

        # الاحتفاظ بآخر 50 رسالة فقط
        data["history"] = data["history"][-50:]

        self._save(data)

    def get_context(self):

        data = self._load()

        return {
            "facts": data.get(
                "facts",
                []
            ),

            "preferences": data.get(
                "preferences",
                []
            ),

            "history": data.get(
                "history",
                []
            )
        }

    def clear_history(self):

        data = self._load()

        data["history"] = []

        self._save(data)

from pathlib import Path
import subprocess
import sys


class FileSystemTools:
    """
    أدوات الملفات الخاصة بـ ALIX.

    كل عمليات الملفات محصورة داخل:
        ~/ALIX-Agent/workspace
    """

    MAX_READ_SIZE = 2 * 1024 * 1024
    MAX_WRITE_SIZE = 2 * 1024 * 1024
    COMMAND_TIMEOUT = 60

    def __init__(self, policy):
        self.policy = policy

    # ========================================================
    # PATH
    # ========================================================

    def _path(self, path):
        """
        تحويل المسار والتحقق من أنه داخل workspace.
        """

        resolved = self.policy.resolve_path(path)

        if resolved is None:
            raise PermissionError(
                "المسار خارج مساحة ALIX."
            )

        return resolved

    # ========================================================
    # LIST
    # ========================================================

    def list_files(
        self,
        path=".",
        all=False
    ):
        base = self._path(path)

        if not base.exists():

            return {
                "ok": False,
                "error": "المسار غير موجود."
            }

        if not base.is_dir():

            return {
                "ok": False,
                "error": "المسار ليس مجلدًا."
            }

        items = []

        try:

            entries = sorted(
                base.iterdir(),
                key=lambda x: (
                    not x.is_dir(),
                    x.name.lower()
                )
            )

            for item in entries:

                if (
                    not all
                    and item.name.startswith(".")
                ):
                    continue

                try:

                    item_type = (
                        "directory"
                        if item.is_dir()
                        else "file"
                    )

                    size = (
                        item.stat().st_size
                        if item.is_file()
                        else None
                    )

                except OSError:

                    item_type = "unknown"
                    size = None

                items.append({
                    "name": item.name,
                    "type": item_type,
                    "size": size
                })

        except Exception as e:

            return {
                "ok": False,
                "error": str(e)
            }

        try:

            relative = base.relative_to(
                self.policy.workspace
            )

            relative_path = (
                str(relative)
                if str(relative) != "."
                else "."
            )

        except ValueError:

            relative_path = "."

        return {
            "ok": True,
            "path": relative_path,
            "items": items[:500]
        }

    # ========================================================
    # READ
    # ========================================================

    def read_file(self, path):

        base = self._path(path)

        if not base.exists():

            return {
                "ok": False,
                "error": "الملف غير موجود."
            }

        if not base.is_file():

            return {
                "ok": False,
                "error": "المسار ليس ملفًا."
            }

        try:

            size = base.stat().st_size

        except OSError as e:

            return {
                "ok": False,
                "error": str(e)
            }

        if size > self.MAX_READ_SIZE:

            return {
                "ok": False,
                "error": (
                    f"الملف أكبر من الحد المسموح "
                    f"({self.MAX_READ_SIZE} bytes)."
                )
            }

        try:

            content = base.read_text(
                encoding="utf-8",
                errors="replace"
            )

            return {
                "ok": True,
                "path": str(
                    base.relative_to(
                        self.policy.workspace
                    )
                ),
                "content": content,
                "bytes": size
            }

        except Exception as e:

            return {
                "ok": False,
                "error": str(e)
            }

    # ========================================================
    # CREATE DIRECTORY
    # ========================================================

    def create_directory(self, path):

        base = self._path(path)

        try:

            base.mkdir(
                parents=True,
                exist_ok=True
            )

            return {
                "ok": True,
                "path": str(
                    base.relative_to(
                        self.policy.workspace
                    )
                )
            }

        except Exception as e:

            return {
                "ok": False,
                "error": str(e)
            }

    # ========================================================
    # WRITE
    # ========================================================

    def write_file(
        self,
        path,
        content
    ):

        base = self._path(path)

        if not isinstance(
            content,
            str
        ):

            return {
                "ok": False,
                "error": "content يجب أن يكون نصًا."
            }

        encoded = content.encode(
            "utf-8"
        )

        if len(encoded) > self.MAX_WRITE_SIZE:

            return {
                "ok": False,
                "error": (
                    f"المحتوى أكبر من الحد المسموح "
                    f"({self.MAX_WRITE_SIZE} bytes)."
                )
            }

        try:

            base.parent.mkdir(
                parents=True,
                exist_ok=True
            )

            base.write_text(
                content,
                encoding="utf-8"
            )

            return {
                "ok": True,
                "path": str(
                    base.relative_to(
                        self.policy.workspace
                    )
                ),
                "bytes": len(encoded)
            }

        except Exception as e:

            return {
                "ok": False,
                "error": str(e)
            }

    # ========================================================
    # DELETE
    # ========================================================

    def delete_file(self, path):

        base = self._path(path)

        if base == self.policy.workspace:

            return {
                "ok": False,
                "error": (
                    "لا يمكن حذف workspace."
                )
            }

        if not base.exists():

            return {
                "ok": False,
                "error": "الملف غير موجود."
            }

        if not base.is_file():

            return {
                "ok": False,
                "error": (
                    "يمكن حذف الملفات فقط، "
                    "وليس المجلدات."
                )
            }

        try:

            base.unlink()

            return {
                "ok": True,
                "deleted": str(
                    base.relative_to(
                        self.policy.workspace
                    )
                )
            }

        except Exception as e:

            return {
                "ok": False,
                "error": str(e)
            }

    # ========================================================
    # RUN PYTHON
    # ========================================================

    def run_python(
        self,
        path,
        args=None
    ):

        base = self._path(path)

        if base.suffix.lower() != ".py":

            return {
                "ok": False,
                "error": (
                    "يجب أن يكون الملف "
                    "Python (.py)."
                )
            }

        if not base.is_file():

            return {
                "ok": False,
                "error": "ملف Python غير موجود."
            }

        if args is None:
            args = []

        if not isinstance(
            args,
            list
        ):

            return {
                "ok": False,
                "error": (
                    "args يجب أن تكون قائمة."
                )
            }

        safe_args = []

        for value in args:

            if not isinstance(
                value,
                (str, int, float)
            ):

                return {
                    "ok": False,
                    "error": (
                        "كل عنصر في args "
                        "يجب أن يكون قيمة بسيطة."
                    )
                }

            safe_args.append(
                str(value)
            )

        command = [
            sys.executable,
            str(base)
        ]

        command.extend(
            safe_args
        )

        try:

            result = subprocess.run(
                command,
                cwd=str(
                    self.policy.workspace
                ),
                stdin=subprocess.DEVNULL,
                capture_output=True,
                text=True,
                timeout=self.COMMAND_TIMEOUT
            )

            return {
                "ok": result.returncode == 0,
                "exit_code": result.returncode,
                "stdout": result.stdout[-12000:],
                "stderr": result.stderr[-12000:]
            }

        except subprocess.TimeoutExpired:

            return {
                "ok": False,
                "error": (
                    "انتهت مهلة تشغيل Python."
                )
            }

        except Exception as e:

            return {
                "ok": False,
                "error": str(e)
            }

import os
import platform
import shutil
import subprocess


class SystemTools:
    """
    أدوات النظام الخاصة بـ ALIX.

    لا يوجد تنفيذ Shell مفتوح.
    أوامر shell تمر أولاً عبر Policy.
    """

    def __init__(self, policy=None):
        self.policy = policy

    # ========================================================
    # SYSTEM INFO
    # ========================================================

    def system_info(self):

        try:
            disk = shutil.disk_usage("/")

            return {
                "ok": True,
                "platform": platform.platform(),
                "system": platform.system(),
                "release": platform.release(),
                "machine": platform.machine(),
                "processor": platform.processor(),
                "python": platform.python_version(),
                "cpu_count": os.cpu_count(),
                "home": str(
                    os.path.expanduser("~")
                ),
                "disk_total": disk.total,
                "disk_used": disk.used,
                "disk_free": disk.free,
            }

        except Exception as e:

            return {
                "ok": False,
                "error": str(e)
            }

    # ========================================================
    # RUN COMMAND
    # ========================================================

    def run_command(
        self,
        command,
        timeout=30
    ):

        if self.policy is None:

            return {
                "ok": False,
                "error": (
                    "Policy غير متوفرة."
                )
            }

        if not isinstance(
            command,
            str
        ):

            return {
                "ok": False,
                "error": (
                    "command يجب أن يكون نصًا."
                )
            }

        command = command.strip()

        if not command:

            return {
                "ok": False,
                "error": "الأمر فارغ."
            }

        # ----------------------------------------------------
        # Security check
        # ----------------------------------------------------

        if not self.policy.command_allowed(
            command
        ):

            return {
                "ok": False,
                "error": (
                    "تم رفض الأمر بواسطة "
                    "سياسة الأمان."
                )
            }

        # ----------------------------------------------------
        # Timeout
        # ----------------------------------------------------

        try:

            timeout = int(timeout)

        except Exception:

            timeout = 30

        timeout = max(
            1,
            min(timeout, 60)
        )

        # ----------------------------------------------------
        # Execute
        # ----------------------------------------------------

        try:

            result = subprocess.run(
                command,
                shell=True,
                cwd=str(
                    self.policy.workspace
                ),
                stdin=subprocess.DEVNULL,
                capture_output=True,
                text=True,
                timeout=timeout
            )

            return {
                "ok": result.returncode == 0,
                "exit_code": result.returncode,
                "stdout": result.stdout[-12000:],
                "stderr": result.stderr[-12000:]
            }

        except subprocess.TimeoutExpired:

            return {
                "ok": False,
                "error": (
                    f"انتهت المهلة بعد "
                    f"{timeout} ثانية."
                )
            }

        except Exception as e:

            return {
                "ok": False,
                "error": str(e)
            }

    # ========================================================
    # GIT STATUS
    # ========================================================

    def git_status(self, path="."):

        if self.policy is None:

            return {
                "ok": False,
                "error": (
                    "Policy غير متوفرة."
                )
            }

        target = self.policy.resolve_path(
            path
        )

        if target is None:

            return {
                "ok": False,
                "error": (
                    "المسار خارج مساحة ALIX."
                )
            }

        if not target.exists():

            return {
                "ok": False,
                "error": (
                    "المسار غير موجود."
                )
            }

        try:

            result = subprocess.run(
                [
                    "git",
                    "-C",
                    str(target),
                    "status",
                    "--short",
                    "--branch"
                ],
                cwd=str(
                    self.policy.workspace
                ),
                stdin=subprocess.DEVNULL,
                capture_output=True,
                text=True,
                timeout=15
            )

            return {
                "ok": result.returncode == 0,
                "exit_code": result.returncode,
                "stdout": result.stdout[-12000:],
                "stderr": result.stderr[-4000:]
            }

        except subprocess.TimeoutExpired:

            return {
                "ok": False,
                "error": (
                    "انتهت مهلة Git."
                )
            }

        except Exception as e:

            return {
                "ok": False,
                "error": str(e)
            }
